pipeline {
    agent any

    environment {

        JAVA_HOME = 'C:\\Program Files\\Java\\jdk-21'

        MAVEN = 'D:\\Softwares\\apache-maven-3.9.4\\bin\\mvn.cmd'

        PATH = "${JAVA_HOME}\\bin;${env.PATH}"
    }

    stages {
        stage('Clone Repository') {
            steps {
                git url: 'https://github.com/SawanPanwar/ORSProject04.git', branch: 'main'
            }
        }

        stage('Build New WAR') {
            steps {
                bat '"%MAVEN%" clean package'
            }
        }

        stage('Deploy Using Docker Compose') {
            steps {
                    bat 'docker-compose down || docker-compose.exe down'
                    bat 'docker-compose up -d --build || docker-compose.exe up -d --build'
            }
        }
    }
}
